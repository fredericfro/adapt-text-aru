            async function handleFile(file) {

                const fileType = file.name.split('.').pop().toLowerCase();

                showStatus(`Lecture de "${file.name}"...`);


                try {

                    if (fileType === 'docx') {

                        // Traitement Word avec Mammoth

                        const arrayBuffer = await file.arrayBuffer();

                        const result = await mammoth.convertToHtml({ arrayBuffer: arrayBuffer });

                        aruDocument.innerHTML = result.value || "<p>Document vide.</p>";

                        hideStatus();

                        applyAnswerSpaces();

                    } else if (fileType === 'pdf') {

                        // Traitement PDF via PDF.js

                        showStatus("Extraction du texte du fichier PDF...");

                        const arrayBuffer = await file.arrayBuffer();

                        const pdf = await pdfjsLib.getDocument({ data: arrayBuffer }).promise;

                        let textHtml = '';


                        for (let i = 1; i <= pdf.numPages; i++) {

                            const page = await pdf.getPage(i);

                            const content = await page.getTextContent();

                            const pageStrings = content.items.map(item => item.str);

                            textHtml += `<p style="margin-bottom:1rem;">${pageStrings.join(' ')}</p>`;

                        }


                        aruDocument.innerHTML = textHtml || "<p>Aucun texte extractible trouvé dans le PDF.</p>";

                        hideStatus();

                        applyAnswerSpaces();

                    } else if (['jpg', 'jpeg', 'png'].includes(fileType)) {

                        // Traitement Image via OCR Tesseract.js

                        showStatus("Analyse de l'image (OCR Tesseract) en cours...");

                        const worker = await Tesseract.createWorker('fra');

                        const ret = await worker.recognize(file);

                        await worker.terminate();


                        const formattedText = ret.data.text.split('\n').filter(line => line.trim() !== '').map(l => `<p>${l}</p>`).join('');

                        aruDocument.innerHTML = formattedText || "<p>Aucun texte détecté dans l'image.</p>";

                        hideStatus();

                        applyAnswerSpaces();

                    } else {

                        alert("Format non pris en charge. Veuillez fournir un fichier .docx, .pdf, .png ou .jpg.");

                        hideStatus();

                    }

                } catch (err) {

                    console.error(err);

                    alert("Erreur lors de la lecture du fichier : " + err.message);

                    hideStatus();

                }

                updateCharCount();

            }

            async function exportToWord() {

                const { Document, Packer, Paragraph, TextRun, BorderStyle } = window.docx;


                const paragraphs = Array.from(aruDocument.children);

                const docChildren = [];


                paragraphs.forEach(node => {

                    const text = node.innerText || '';

                    const isHeader = node.tagName === 'H1' || node.tagName === 'H2';


                    if (node.classList.contains('aru-response-container')) {

                        // Lignes de réponse Word

                        const lineCount = node.children.length;

                        for(let i=0; i<lineCount; i++) {

                            docChildren.push(new Paragraph({

                                border: { bottom: { color: "94A3B8", space: 1, value: BorderStyle.DASHED, size: 6 } },

                                spacing: { after: 200, before: 200 }

                            }));

                        }

                    } else if (node.classList.contains('aru-response-box')) {

                        // Cadre de réponse

                        docChildren.push(new Paragraph({

                            text: "[ Zone réservée aux réponses de l'élève ]",

                            border: {

                                top: { color: "CBD5E1", value: BorderStyle.SINGLE, size: 6 },

                                bottom: { color: "CBD5E1", value: BorderStyle.SINGLE, size: 6 },

                                left: { color: "CBD5E1", value: BorderStyle.SINGLE, size: 6 },

                                right: { color: "CBD5E1", value: BorderStyle.SINGLE, size: 6 },

                            },

                            spacing: { before: 200, after: 200 }

                        }));

                    } else if (text.trim() !== '') {

                        docChildren.push(new Paragraph({

                            children: [

                                new TextRun({

                                    text: text,

                                    size: parseInt(fontSizeInput.value) * 2, // docx utilise demi-points (14pt = 28)

                                    font: "Arial",

                                    bold: isHeader || node.style.fontWeight === 'bold'

                                })

                            ],
